let map; let markers = []; // 생성된 마커들을 관리할 배열 // 1. 네이버 지도 초기화 function initMap() { const mapOptions = { center: new naver.maps.LatLng(37.555142, 126.970447), // 기본 중심지: 서울역 인근 zoom: 13, zoomControl: true, zoomControlOptions: { position: naver.maps.Position.RIGHT_CENTER } }; map = new naver.maps.Map('map-container', mapOptions); } if (typeof naver !== 'undefined' && naver.maps) { initMap(); } else { console.error("네이버 지도 스크립트 로드 실패"); } // 플로팅 버튼 클릭 시 파일 선택창 트리거 function triggerUpload() { document.getElementById('file-input').click(); } // 2. 사진을 선택했을 때 처리하는 핵심 비즈니스 로직 function handleFiles(files) { if (files.length === 0) return; Array.from(files).forEach((file) => { // 브라우저에서 사용할 임시 이미지 URL 생성 const imageUrl = URL.createObjectURL(file); // EXIF 데이터 추출 시작 EXIF.getData(file, function() { // 위도(Lat), 경도(Lng) 가져오기 const lat = EXIF.getTag(this, "GPSLatitude"); const lng = EXIF.getTag(this, "GPSLongitude"); const latRef = EXIF.getTag(this, "GPSLatitudeRef") || "N"; const lngRef = EXIF.getTag(this, "GPSLongitudeRef") || "E"; // GPS 정보가 유효하게 존재하는 경우 if (lat && lng) { // EXIF 내부의 도/분/초(도분초 도 단위 변환) 형태 포맷 변환하기 const finalLat = convertToDecimal(lat, latRef); const finalLng = convertToDecimal(lng, lngRef); // 변환된 실제 위치에 사진 마커 생성 createPhotoMarker(finalLat, finalLng, imageUrl); } else { // [예외 처리] 사진에 GPS 정보가 없는 경우 고지 후 현재 지도 중심 근처에 뿌림 console.warn(`${file.name} 사진에 GPS 정보가 없습니다. 현재 중심 위치에 임시 표시합니다.`); const currentCenter = map.getCenter(); const deltaLat = (Math.random() - 0.5) * 0.01; const deltaLng = (Math.random() - 0.5) * 0.01; createPhotoMarker(currentCenter.lat() + deltaLat, currentCenter.lng() + deltaLng, imageUrl); } }); }); // input 초기화 document.getElementById('file-input').value = ''; } // 3. EXIF 도/분/초 데이터를 지도 인식용 십진수(Decimal) 좌표로 변환하는 함수 function convertToDecimal(gpsData, ref) { // gpsData = [도, 분, 초] 배열 형태임 const degrees = gpsData[0]; const minutes = gpsData[1]; const seconds = gpsData[2]; let decimal = degrees + (minutes / 60) + (seconds / 3600); // 남위(S)나 서경(W)일 경우 음수로 변환 (한국은 북위N, 동경E라 보통 그대로 양수) if (ref === "S" || ref === "W") { decimal = decimal * -1; } return decimal; } // 4. 지도 위에 커스텀 이미지 마커를 생성하는 함수 function createPhotoMarker(lat, lng, imageUrl) { const position = new naver.maps.LatLng(lat, lng); const markerContent = `
`; const marker = new naver.maps.Marker({ position: position, map: map, icon: { content: markerContent, anchor: new naver.maps.Point(27.5, 27.5) } }); markers.push(marker); // [선택 사항] 새로 추가된 사진의 위치로 지도 중심 이동하기 map.setCenter(position); }
